\usepackage{tikz}

\begin{document}
\begin{tikzpicture}[
    node distance=2.8cm,
    every node/.style={font=\sffamily\bfseries, text=teal!95!black},
    process/.style={rectangle, draw=blue!80!black, fill=cyan!10, line width=1.2pt, minimum height=1cm, minimum width=2.7cm, rounded corners=3pt},
    startstop/.style={ellipse, draw=teal!90!black, fill=blue!10, line width=1.2pt, minimum height=1cm, minimum width=2.5cm},

\draw[arrow] (compile) -- (reference);
\draw[arrow] (reference) -- (index);
