% Custom Voltage Arrows
\newcommand{\fixedvlen}[3][0.54cm]{% [semilength]{node}{label}
    \coordinate (#2-Vcenter) at ($(#2-Vfrom)!0.5!(#2-Vto)$);
    \draw[-{Stealth[scale=1.1]}, color=blue!80!black] ($(#2-Vcenter)!#1!(#2-Vfrom)$) -- ($(#2-Vcenter)!#1!(#2-Vto)$);
    \node[ anchor=\ctikzgetanchor{#2}{Vlab}, text=blue!90!black] at (#2-Vlab) {#3};
}
\newcommand{\fixedvlendashed}[3][0.8cm]{% [semilength]{node}{label}
    \coordinate (#2-Vcenter) at ($(#2-Vfrom)!0.5!(#2-Vto)$);
    \draw[dashed,-{Stealth[scale=1]}, color=teal!80!black] ($(#2-Vcenter)!#1!(#2-Vfrom)$) -- ($(#2-Vcenter)!#1!(#2-Vto)$);
    \node[ anchor=\ctikzgetanchor{#2}{Vlab}, text=teal!90!black] at (#2-Vlab) {#3};
}
