\tikzset{ADDER/.style={fill=blue!25!black,draw=teal,very thick,minimum size=15pt,circle,append after command={
        [shorten >=\pgflinewidth, shorten <=\pgflinewidth,]
        (\tikzlastnode.north) edge[teal,thick] (\tikzlastnode.south)
        (\tikzlastnode.east) edge[teal,thick] (\tikzlastnode.west)
        }
    }
}

\tikzstyle{process}=[draw,fill=blue!15!black,rectangle,rounded corners=2pt,minimum size=22pt,inner sep=2pt,thick,teal]
\begin{tikzpicture}
    \newcommand{\n}{3}
    \foreach \nr in {1, ..., \n}{
        \node (D\nr)            at (0,{(\n-\nr)*3}) {\texttt{Block$_\nr$}};
        \node (A\nr)[ADDER]     at (2,{(\n-\nr)*3}) {};
        \node (P\nr)[process]   at (2,{(\n-\nr)*3+1}) {\texttt{Func}};
        \node (Key\nr)          at (2,{(\n-\nr)*3+2}) {\texttt{Key$_\nr$}};
        \node (O\nr)            at (4,{(\n-\nr)*3}) {$S_\nr = 2B_\nr - 3$};

        \draw[-stealth,thick,teal] (D\nr) -- (A\nr);
        \draw[-stealth,thick,teal] (A\nr) -- (O\nr);
        \draw[-stealth,thick,teal] (Key\nr) -- (P\nr);
        \draw[-stealth,thick,teal] (P\nr) -- (A\nr);
    }

    \foreach \nr in {2, ..., \n}{
        \pgfmathtruncatemacro{\tmp}{\nr-1}
        \draw[-stealth,thick,teal] (2, {(\n-\tmp)*3+0.5}) -- (1, {(\n-\tmp)*3+0.5}) -- (1, {(\n-\nr)*3+1}) -- (P\nr);
    }

    \node (IV) at (0.5,{\n*3-2}) {\texttt{Init}};
    \draw[-stealth,thick,teal] (IV) -- (P1);

    \node (Dx) at (0,-1) {$\vdots$};
    \node (Ax) at (2,-1) {$\vdots$};
    \node (Ox) at (4,-1) {$\vdots$};
