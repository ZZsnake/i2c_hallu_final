\documentclass[border=10pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{calc}

\begin{document}
\begin{tikzpicture}[scale=3,
  declare function={
    % simple surface: phi(x,y)=0.2*(x^2 + y^2) - 0.4*x*y
    phisurf(\x,\y) = 0.2*(\x*\x+\y*\y) -0.4*\x*\y;
  },
  line join=round, line cap=round]

%--- 3D to 2D projection macros (oblique view) ---
% x axis: to the right and slightly down
% y axis: to the right and slightly up
% z axis: up
\def\xx{0.9}  \def\xy{-0.15}
\def\yx{-0.6} \def\yy{-0.15}
\def\zx{0}    \def\zy{0.8}

\newcommand{\Pthree}[3]{%
  % projects (X,Y,Z) -> (u,v)
  ({#1*\xx + #2*\yx + #3*\zx},
   {#1*\xy + #2*\yy + #3*\zy})}

% Colors
\colorlet{framecol}{teal!80!black}
\colorlet{surfcol}{blue!80!black}

% Box size
\def\Xmax{1.2}
\def\Ymax{1.2}
\def\Zmax{1.0}

%----------------- draw 3D box -------------------
\begin{scope}[line width=0.6pt, framecol]
  % bottom rectangle
  \draw[framecol]
    \Pthree{0}{0}{0} --
    \Pthree{\Xmax}{0}{0} --
    \Pthree{\Xmax}{\Ymax}{0} --
    \Pthree{0}{\Ymax}{0} --
    cycle;
  % vertical edges
  \draw[framecol]
    \Pthree{0}{0}{0} -- \Pthree{0}{0}{\Zmax};
  \draw[framecol]
    \Pthree{\Xmax}{0}{0} -- \Pthree{\Xmax}{0}{\Zmax};
  \draw[framecol]
    \Pthree{0}{\Ymax}{0} -- \Pthree{0}{\Ymax}{\Zmax};
  \draw[framecol]
    \Pthree{\Xmax}{\Ymax}{0} -- \Pthree{\Xmax}{\Ymax}{\Zmax};
  % top rectangle
  \draw[framecol]
    \Pthree{0}{0}{\Zmax} --
    \Pthree{\Xmax}{0}{\Zmax} --
    \Pthree{\Xmax}{\Ymax}{\Zmax} --
    \Pthree{0}{\Ymax}{\Zmax} --
    cycle;
\end{scope}

%----------------- surface grid -------------------
\begin{scope}
  \clip
    \Pthree{0}{0}{0} --
    \Pthree{\Xmax}{0}{0} --
    \Pthree{\Xmax}{\Ymax}{0} --
    \Pthree{0}{\Ymax}{0} --
    cycle;
\end{scope}

% To ensure the surface is inside the box and visible, draw without clipping the top.
\begin{scope}
  % fill underlaid surface for solid impression (optional light fill)
  \path[fill=surfcol!40]
    \foreach \xa [evaluate=\xa as \ya using 0] in {0,0.05,...,1.2} {
      % dummy, closed later
    };
\end{scope}

\begin{scope}[surfcol, line width=0.25pt]
  % grid in xi-direction (constant eta)
  \foreach \eta in {0,0.05,...,1.2} {
    \draw[surfcol]
      \foreach \xi in {0,0.05,...,1.2} {
        \ifnum\pdfstrcmp{\xi}{0}=0
          \Pthree{\xi}{\eta}{phisurf(\xi,\eta)}
        \else
          -- \Pthree{\xi}{\eta}{phisurf(\xi,\eta)}
        \fi
      };
  }
  % grid in eta-direction (constant xi)
  \foreach \xi in {0,0.05,...,1.2} {
    \draw[surfcol]
      \foreach \eta in {0,0.05,...,1.2} {
        \ifnum\pdfstrcmp{\eta}{0}=0
          \Pthree{\xi}{\eta}{phisurf(\xi,\eta)}
        \else
          -- \Pthree{\xi}{\eta}{phisurf(\xi,\eta)}
        \fi
      };
  }
\end{scope}

%----------------- axis labels -------------------
\begin{scope}[framecol]
  % xi label along bottom left edge
  \node[anchor=north] at ($( \Pthree{0}{0}{0} !0.52! \Pthree{\Xmax}{0}{0} )$)
    {\large$\xi$};

  % eta label along bottom right edge
  \node[anchor=north east] at
    ($( \Pthree{\Xmax}{0}{0} !0.55! \Pthree{\Xmax}{\Ymax}{0} )$)
    {\large$\eta$};

  % Phi(xi,eta) label on vertical axis, rotated
  \node[anchor=south, rotate=90] at
    ($( \Pthree{0}{0}{0} !0.5! \Pthree{0}{0}{\Zmax} )$)
    {\large$\Phi(\xi,\eta)$};
\end{scope}

\end{tikzpicture}
\end{document}
