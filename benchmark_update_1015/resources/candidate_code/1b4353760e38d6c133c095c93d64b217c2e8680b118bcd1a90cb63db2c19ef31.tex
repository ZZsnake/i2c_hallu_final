% Define 7-segment display style
\tikzset{
  segment/.style={thick, red},
  segment-gray/.style={thick, gray},
  segment-green/.style={thick, green},
  box/.style={draw=blue, thick, minimum size=0.8cm, inner sep=0pt}
}

% Define coordinates for 7-segment display
\newcommand{\drawSegment}[2]{
  \draw[#1] (#2) -- ++(0.8,0);
}

% Define 7-segment display with segments labeled a-g
\newcommand{\sevenSegment}[1]{
  \begin{scope}[shift={(0,0)}]
    \draw[segment] (0,0.8) -- (0.8,0.8); % a
    \draw[segment] (0.8,0.8) -- (0.8,0.4); % b
    \draw[segment] (0.8,0.4) -- (0.8,0); % c
    \draw[segment] (0,0) -- (0.8,0); % d
    \draw[segment] (0,0) -- (0,0.4); % e
    \draw[segment] (0,0.4) -- (0,0.8); % f
    \draw[segment] (0.4,0.8) -- (0.4,0.4); % g
    \draw[segment] (0.4,0.4) -- (0.4,0); % g (continued)
  \end{scope}
}
