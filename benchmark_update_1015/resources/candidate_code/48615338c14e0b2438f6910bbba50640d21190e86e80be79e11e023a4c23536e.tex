\documentclass[border=10pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{3d,calc}

\begin{document}
\begin{tikzpicture}[scale=1.5, >=Stealth,
    x={(1cm,0cm)}, y={(0.707cm,0.707cm)}, z={(0cm,1cm)},
    axis/.style={->,thick},
    surface/.style={cyan!50, opacity=0.7},
    volume/.style={cyan, thick},
    plane/.style={gray, fill=gray!30},
    label/.style={font=\small},
]

% Axes
\draw[axis] (0,0,0) -- (4,0,0) node[right] {$x$};
\draw[axis] (0,0,0) -- (0,4,0) node[above right] {$y$};
\draw[axis] (0,0,0) -- (0,0,4) node[above] {$z$};

% Surface
\begin{scope}[canvas is xy plane at z=3]
    \draw[surface] (0,0) grid (4,4);
    \draw[blue, thick] (1,1) rectangle (3,3);
\end{scope}

% Volume element dV
\draw[volume] (2,2,0) -- (2,2,3) node[midway, right] {$dV$};

% Rectangular region R in xy-plane
\draw[plane] (1,1,0) -- (3,1,0) -- (3,3,0) -- (1,3,0) -- cycle;
\node[label] at (2,2,-0.5) {$R$};

% Labels for vertices of R
\foreach \point/\position in {(1,1,0)/left, (3,1,0)/right, (3,3,0)/right, (1,3,0)/left}
    \node[label] at \point {\position: $a$};

% Mathematical notation
\node[label] at (2,2,3.5) {$z = f(x, y)$};

\end{tikzpicture}
\end{document}
